    private var connectionStatus by mutableStateOf("OFF")
    private var subscriptionLabel by mutableStateOf("Проверяем подписку…")
    private fun startHealthPolling() {
        subscriptionLabel=SofixSubscription.cached(this)
        lifecycleScope.launch {
            var count=0
            while (!isFinishing) {
                val running=mainViewModel.uiState.value.isRunning
                val quick=SofixHealth.fast(this@MainActivity,running)
                if(quick!="CHECKING" || connectionStatus=="OFF")connectionStatus=quick
                if (running && SofixHealth.physicalAvailable(this@MainActivity) && count%3==0) {
                    val sampled=withContext(Dispatchers.IO){SofixHealth.sample(this@MainActivity,true)}
                    connectionStatus=sampled.name
                }
                if(count%3==0) {
                    subscriptionLabel=if(legacyRevoked) {
                        "Телефон отвязан · восстановите через Telegram"
                    } else try {withContext(Dispatchers.IO){
                        SofixSubscription.fetch(this@MainActivity)
                    }}catch (_:Exception){
                        SofixSubscription.cached(this@MainActivity)+" · последняя проверка"
                    }
                }
                if(count%180==0) {
                    withContext(Dispatchers.IO){
                        try {SofixRuleUpdates.refresh(this@MainActivity)}catch (_:Exception){}
                    }
                }
                count++
                delay(20_000)
            }
        }
    }

    private var offerTrial by mutableStateOf(false)
    private var trialMinutesLeft by mutableIntStateOf(0)
    private var trialBusy by mutableStateOf(false)
    private var trialHint by mutableStateOf("")
    private var trialLinked by mutableStateOf(false)
    private var trialPaid by mutableStateOf(false)
    private var trialTelegramOpened = false
    private var linkCheckBusy = false

    private fun updateTrialClock() {
        trialMinutesLeft = ((SofixTrialManager.expiresAt(this) - System.currentTimeMillis()/1000 + 59) / 60)
            .toInt().coerceAtLeast(0)
    }

    private fun startTrialPolling() {
        trialLinked = SofixTrialManager.linked(this)
        trialPaid = SofixTrialManager.paid(this)
        updateTrialClock()
        lifecycleScope.launch {
            while (!isFinishing) {
                updateTrialClock()
                delay(10_000)
            }
        }
    }

    private var legacyRevoked by mutableStateOf(false)
    private var paidPairBusy by mutableStateOf(false)
    private var showPaidPairDialog by mutableStateOf(false)
    private var paidPairPin by mutableStateOf("")

    override fun onResume() {
        super.onResume()
        updateTrialClock()
        // Pending paid pairing has priority after returning from Telegram.
        // An old anonymous trial must never mask a recovered paid device.
        when {
            SofixPaidPair.isPending(this) -> resumePaidPair()
            SofixLegacyPair.token(this)!=null -> refreshLegacyDeviceState()
            SofixTrialManager.token(this)!=null -> refreshTrialServerState()
        }
    }

    /** Import and select the paid device's OWN key, not a revoked older key. */
    private suspend fun installVerifiedPaidConfig(url:String) {
        require(url.startsWith("https://"))
        var sub=MmkvManager.decodeSubscriptions().firstOrNull { it.subscription.url==url }
        if(sub==null) {
            val added=withContext(Dispatchers.IO){
                AngConfigManager.importBatchConfig(url,"",false)
            }
            if(added.first+added.second<=0)
                throw IllegalStateException("Не удалось загрузить новую VPN-конфигурацию")
            sub=MmkvManager.decodeSubscriptions().firstOrNull { it.subscription.url==url }
        }
        val verifiedSub=sub ?: throw IllegalStateException("VPN-подписка не сохранилась")
        var candidates=MmkvManager.decodeServerList(verifiedSub.guid)
        if(candidates.isEmpty()) {
            withContext(Dispatchers.IO){AngConfigManager.updateConfigViaSub(verifiedSub)}
            candidates=MmkvManager.decodeServerList(verifiedSub.guid)
        }
        val selected=candidates.firstOrNull()
            ?: throw IllegalStateException("Не удалось получить сервер VPN. Повторите проверку.")
        mainViewModel.onAction(MainAction.SelectServer(selected))
        mainViewModel.onAction(MainAction.RefreshGroups)
    }

    /** Existing paid customer: no free trial, no new payment, no automatic
     * reactivation of revoked tokens. The owner must confirm in Telegram. */
    private fun beginPaidPair() {
        if(paidPairBusy)return
        if(SofixPaidPair.isPending(this)) {
            paidPairPin=SofixPaidPair.pin(this).orEmpty()
            showPaidPairDialog=true
            return
        }
        paidPairBusy=true
        trialHint="Готовим безопасную привязку телефона…"
        lifecycleScope.launch {
            try {
                withContext(Dispatchers.IO){SofixPaidPair.start(this@MainActivity)}
                paidPairPin=SofixPaidPair.pin(this@MainActivity).orEmpty()
                showPaidPairDialog=true
                trialHint="Подтвердите этот телефон в своём Telegram. Повторная оплата не нужна."
            }catch(e:Exception){
                trialHint="Не удалось начать привязку: "+(e.message?:"Проверьте интернет")
            }finally {paidPairBusy=false}
        }
    }

    private fun openPaidPairTelegram() {
        val link=SofixPaidPair.telegramUrl(this) ?: run {
            trialHint="Запрос устарел. Создайте новый код привязки."
            return
        }
        try {
            startActivity(Intent(Intent.ACTION_VIEW,Uri.parse(link))
                .addCategory(Intent.CATEGORY_BROWSABLE))
            showPaidPairDialog=false
        }catch (_:Exception){
            trialHint="Не удалось открыть Telegram. Проверьте приложение Telegram."
        }
    }

    private fun resumePaidPair() {
        if(paidPairBusy || !SofixPaidPair.isPending(this))return
        paidPairBusy=true
        lifecycleScope.launch {
            try {
                val status=withContext(Dispatchers.IO){SofixPaidPair.check(this@MainActivity)}
                when(status.optString("state")) {
                    "ready" -> {
                        val url=status.getString("subscription_url")
                        installVerifiedPaidConfig(url)
                        // The token is moved only after server proof and config import.
                        SofixPaidPair.complete(this@MainActivity)
                        legacyRevoked=false
                        showPaidPairDialog=false
                        trialHint="Подписка и этот телефон успешно привязаны."
                        subscriptionLabel=withContext(Dispatchers.IO){
                            SofixSubscription.fetch(this@MainActivity)
                        }
                    }
                    "pending" -> trialHint="Подтвердите привязку телефона в Telegram, указав код с экрана."
                    "setting_up" -> trialHint="Telegram подтвердил телефон. Сервер готовит VPN-ключ. Повторите проверку."
                    "no_subscription" -> trialHint="Телефон подтверждён, но у Telegram-аккаунта нет активной подписки."
                    else -> trialHint="Проверяем привязку…"
                }
            }catch(e:SofixApiException) {
                if(e.statusCode==401 || e.statusCode==403) {
                    // Keep the old paid credential unchanged. A new challenge
                    // may be started, but only by an explicit user action.
                    SofixPaidPair.clear(this@MainActivity)
                    showPaidPairDialog=false
                    trialHint="Код привязки недействителен. Нажмите «Восстановить подписку» для нового кода."
                }else trialHint="Проверка привязки недоступна. Повторите после восстановления интернета."
            }catch(e:Exception) {
                trialHint="Не удалось закончить привязку: "+(e.message?:"Повторите проверку")
            }finally {paidPairBusy=false}
        }
    }

    private var legacySyncInProgress = false

    /**
     * A paid legacy device can already be active on the server while the old
     * local pairing flow never wrote pair_complete. Verify the saved token
     * and restore its subscription without asking for another payment.
     */
    private fun refreshLegacyDeviceState() {
        if(legacySyncInProgress)return
        val token=SofixLegacyPair.token(this) ?: return
        legacySyncInProgress=true
        lifecycleScope.launch {
            try {
                val result=withContext(Dispatchers.IO) {
                    SofixTrialManager.api("GET","/mobile/api/me",token)
                }
                when(result.optString("state")) {
                    "ready" -> {
                        val url=result.optString("subscription_url")
                        if(!url.startsWith("https://"))
                            throw IllegalStateException("Сервер не вернул подписку")
                        installVerifiedPaidConfig(url)
                        SofixLegacyPair.markConfirmed(this@MainActivity)
                        legacyRevoked=false
                        trialHint=""
                        subscriptionLabel=try { withContext(Dispatchers.IO) {
                            SofixSubscription.fetch(this@MainActivity)
                        }} catch (_:Exception) { "Подписка подтверждена" }
                    }
                    "pending","setting_up" -> trialHint="Привязка телефона завершается. Повторите проверку."
                    "no_subscription" -> trialHint="Телефон привязан, но подписка на сервере неактивна."
                    else -> trialHint="Подписка устройства ещё не подтверждена."
                }
            }catch(e:SofixApiException) {
                if(e.statusCode==401 || e.statusCode==403) {
                    // Actual server rejection is different from a network error.
                    legacyRevoked=true
                    subscriptionLabel="Телефон отвязан · восстановите через Telegram"
                    trialHint="Действующая подписка принадлежит Telegram-аккаунту. "+
                        "Этот телефон нужно привязать заново, платить повторно не нужно."
                }else {
                    trialHint="Не удалось проверить подписку. Проверьте интернет и повторите."
                }
            }catch(e:Exception) {
                // Network failures must not turn a paid subscriber into a trial.
                trialHint="Не удалось проверить подключение: "+
                    (e.message ?: "временная ошибка")
            }finally { legacySyncInProgress=false }
        }
    }

    private fun refreshTrialServerState() {
        lifecycleScope.launch {
            val result=try {
                withContext(Dispatchers.IO){SofixTrialManager.status(this@MainActivity)}
            }catch (_:Exception){null}
            if (result==null) return@launch
            val linked=result.optBoolean("linked",false)
            if (linked){
                SofixTrialManager.markLinked(this@MainActivity)
                trialLinked=true
                if(result.optString("state")=="ready"){
                    val paidUrl=result.optString("subscription_url")
                    if(paidUrl.startsWith("https://") && !trialPaid){
                        withContext(Dispatchers.IO){AngConfigManager.importBatchConfig(paidUrl,"",false)}
                        SofixTrialManager.markPaid(this@MainActivity)
                        trialPaid=true
                        mainViewModel.onAction(MainAction.RefreshGroups)
                    }
                }else trialHint="Телефон привязан к Telegram. Оформите подписку в Mini App."
            }
        }
    }

    private fun showSubscriptionOffer() {
        if (trialPaid || SofixLegacyPair.paired(this)) {
            openSofixMiniApp("renew")
        } else if (SofixTrialManager.token(this)!=null) {
            if (!trialLinked) verifyAndLinkTelegram()
            else openSofixMiniApp("renew")
        } else offerTrial=true
    }

    private fun beginFreeTrial() {
        offerTrial=false
        if (trialBusy) return
        trialBusy=true
        trialHint="Получаем 15 минут бесплатного VPN…"
        lifecycleScope.launch {
            try{
                val payload=withContext(Dispatchers.IO){ SofixTrialManager.start(this@MainActivity) }
                updateTrialClock()
                trialHint="Загружаем VPN-конфигурацию…"
                val imported=withContext(Dispatchers.IO){
                    AngConfigManager.importBatchConfig(payload.getString("subscription_url"),"",false)
                }
                if(imported.first+imported.second < 1)
                    throw IllegalStateException("Не удалось загрузить VPN-конфигурацию")
                mainViewModel.onAction(MainAction.RefreshGroups)
                trialHint="Настраиваем VPN…"
                var ready=false
                for(i in 0 until 25){
                    if(mainViewModel.uiState.value.selectedGuid.isNullOrBlank()) delay(400)
                    else {ready=true;break}
                }
                if(!ready && mainViewModel.uiState.value.selectedGuid.isNullOrBlank())
                    throw IllegalStateException("Не удалось выбрать VPN-сервер")
                requestServiceStart()
                trialHint="Разрешите VPN-подключение в Android. Затем проверим доступ в интернет."
                verifyAndLinkTelegram()
            }catch(e:Exception){
                trialHint="Не удалось включить пробный VPN: "+(e.message?:"Проверьте интернет")
            }finally{trialBusy=false}
        }
    }

    private fun verifyAndLinkTelegram() {
        if(linkCheckBusy || trialLinked) {
            if(trialLinked)openSofixMiniApp("renew")
            return
        }
        linkCheckBusy=true
        lifecycleScope.launch {
            try{
                trialHint="Проверяем VPN и доступность Telegram…"
                var connected=false
                for(i in 0 until 24){
                    if(mainViewModel.uiState.value.isRunning){connected=true;break}
                    delay(1000)
                }
                if(!connected) {
                    trialHint="VPN ещё не подключён. Включите его большой кнопкой и повторите проверку."
                    return@launch
                }
                val ok=withContext(Dispatchers.IO){SofixTrialManager.connectivityCheck()}
                if(!ok){
                    trialHint="VPN включён, но доступность интернета и Telegram не подтверждена. Повторите проверку."
                    return@launch
                }
                trialHint="Интернет и Telegram доступны. Привязываем аккаунт…"
                val telegramUrl=withContext(Dispatchers.IO){SofixTrialManager.beginLink(this@MainActivity)}
                if(!trialTelegramOpened){
                    trialTelegramOpened=true
                    startActivity(Intent(Intent.ACTION_VIEW,Uri.parse(telegramUrl)).addCategory(Intent.CATEGORY_BROWSABLE))
                } else trialHint="Откройте Telegram для подтверждения привязки телефона."
            }catch(e:Exception){
                trialHint="Ошибка привязки: "+(e.message?:"Повторите попытку")
            }finally{linkCheckBusy=false}
        }
    }
